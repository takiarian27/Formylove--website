<script>
  const pages = ["intro", "page1", "page2", "page3", "page4", "page5"];
  let currentPage = 0;
  let clickCount = 0;
  const letter = document.getElementById("letter");

  letter.addEventListener("click", () => {
    clickCount++;
    letter.classList.add("pulse");

    setTimeout(() => letter.classList.remove("pulse"), 200);

    if (clickCount >= 10) {
      letter.classList.add("pop");
      setTimeout(() => {
        showPage(1);
      }, 600);
    }
  });

  function showPage(index) {
    document.querySelectorAll(".page").forEach(p => p.classList.remove("active"));
    document.getElementById(pages[index]).classList.add("active");
    currentPage = index;
  }

  function nextPage() {
    if (currentPage < pages.length - 1) {
      showPage(currentPage + 1);
    }
  }
</script>
